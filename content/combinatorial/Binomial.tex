Some nice properties of $C^k_n$:
\begin{itemize}
    \item $\displaystyle
    C^k_n = C^{n-k}_n = \frac{n}{k} \ C^{n-1}_{k-1}, \sum_{k = 0}^n C^k_n = 2^n.$

    \item $\displaystyle
    \sum_{m = 0}^n C^k_m = C^{n+1}_{k+1}, \sum_{k = 0}^m C^k_{n+k} = C^m_{n+m+1}$
    
    \item $\displaystyle
    \sum_{k = 0}^n (C^k_n)^2 = C^n_{2n}.$
    , $\displaystyle
    \sum_{k = 0}^n k \cdot C^k_n = n2^{n - 1}$

    \item $\displaystyle
    \sum_{k = 0}^n C^k_{n-k} = F_{n + 1}$ (Fibonacci number).
    \item $\displaystyle
    \sum_{k = 0}^{x} \frac{C^k_x}{C^k_n} = \frac{n + 1}{n - x + 1}.$
    
\end{itemize}
